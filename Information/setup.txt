// Production setup --------------------------------------------------------------------------
-- 1️⃣ Connect as PostgreSQL superuser
psql -U postgres -h localhost -p 5432

-- 2️⃣ Create application user (role)
CREATE USER "doctorbook_user"
WITH PASSWORD 'Doctorbook@2026';

-- 3️⃣ Create database and assign ownership
CREATE DATABASE "doctorbook_db"
OWNER "doctorbook_user";

-- 4️⃣ Restrict public access and grant access to application user
REVOKE CONNECT ON DATABASE "doctorbook_db" FROM PUBLIC;
GRANT CONNECT ON DATABASE "doctorbook_db" TO "doctorbook_user";

-- 5️⃣ Connect to the database
\c doctorbook_db

-- 6️⃣ Create schema
CREATE SCHEMA IF NOT EXISTS doctorbook_schema
AUTHORIZATION "doctorbook_user";

-- 7️⃣ Secure schema permissions
REVOKE ALL ON SCHEMA doctorbook_schema FROM PUBLIC;

GRANT USAGE ON SCHEMA doctorbook_schema
TO "doctorbook_user";

GRANT CREATE ON SCHEMA doctorbook_schema
TO "doctorbook_user";

-- 8️⃣ Configure default privileges for future tables
ALTER DEFAULT PRIVILEGES
FOR ROLE "doctorbook_user"
IN SCHEMA doctorbook_schema
GRANT SELECT, INSERT, UPDATE, DELETE
ON TABLES TO "doctorbook_user";

-- Configure default privileges for future sequences
ALTER DEFAULT PRIVILEGES
FOR ROLE "doctorbook_user"
IN SCHEMA doctorbook_schema
GRANT USAGE, SELECT
ON SEQUENCES TO "doctorbook_user";

-- 9️⃣ Set default search path
ALTER ROLE "doctorbook_user"
SET search_path = doctorbook_schema, public;

-- 🔟 Install required extensions
CREATE EXTENSION IF NOT EXISTS pgcrypto;
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

-- 1️⃣1️⃣ Grant additional permissions
GRANT CONNECT ON DATABASE "doctorbook_db"
TO "doctorbook_user";

GRANT USAGE, CREATE ON SCHEMA doctorbook_schema
TO "doctorbook_user";

GRANT ALL PRIVILEGES ON DATABASE "doctorbook_db"
TO "doctorbook_user";

GRANT ALL PRIVILEGES ON ALL TABLES IN SCHEMA doctorbook_schema
TO "doctorbook_user";

GRANT ALL PRIVILEGES ON ALL SEQUENCES IN SCHEMA doctorbook_schema
TO "doctorbook_user";

-- 1️⃣2️⃣ Ensure future objects automatically inherit permissions
ALTER DEFAULT PRIVILEGES
IN SCHEMA doctorbook_schema
GRANT ALL PRIVILEGES ON TABLES
TO "doctorbook_user";

ALTER DEFAULT PRIVILEGES
IN SCHEMA doctorbook_schema
GRANT ALL PRIVILEGES ON SEQUENCES
TO "doctorbook_user";

-- 1️⃣3️⃣ Verify
\du doctorbook_user
\l doctorbook_db
\dn doctorbook_schema

\q

// Local / Development setup --------------------------------------------------------------------------

1️⃣ Connect as superuser
psql -U postgres -h localhost -p 5432
# On this machine postgres password may be: root

2️⃣–1️⃣2️⃣ Run the same SQL as Production (doctorbook_user / doctorbook_db / doctorbook_schema)

3️⃣ Copy env file (from backend-doctorbook folder)
copy .env.example .env

4️⃣ Install + migrate + seed
npm install
npx sequelize-cli db:migrate
npx sequelize-cli db:seed:all

# Or npm scripts:
npm run db:migrate
npm run db:seed

5️⃣ Start API
npm run dev

# Undo last migration
npx sequelize-cli db:migrate:undo

# Undo all migrations
npx sequelize-cli db:migrate:undo:all

# Undo all seeds
npx sequelize-cli db:seed:undo:all

# 2. Deploy new backend code.
